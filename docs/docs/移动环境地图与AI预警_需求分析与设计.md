class Aligner:
    def __init__(self):
        self.ia1_buffer = []       # [(received_ms, payload)]
        self.pending_st1 = []      # [(received_ms, payload)]
        self.lock = RLock()

    def on_ia1(self, payload: dict, received_ms: int):
        with self.lock:
            self.ia1_buffer.append((received_ms, payload))
            # 鍙繚鐣欐渶杩?10 绉?            cutoff = received_ms - 10_000
            self.ia1_buffer = [
                item for item in self.ia1_buffer if item[0] >= cutoff
            ]

    def on_st1(self, payload: dict, received_ms: int):
        if not payload.get("gps_valid"):
            return None
        with self.lock:
            self.pending_st1.append((received_ms, payload))
        return None

    def flush(self, now: int, run_id: int, db) -> list[dict]:
        """姣?200ms 璋冧竴娆★細鎶婄瓑寰呰秴杩?2.5s 鐨?ST1 鐐归厤瀵广€?""
        ready = []
        with self.lock:
            keep = []
            for gps_ms, gps in self.pending_st1:
                if now - gps_ms < 2500:
                    keep.append((gps_ms, gps))
                    continue
                env, delta = self._nearest(gps_ms)
                ready.append({
                    "run_id": run_id,
                    "gps_ms": gps_ms,
                    "lat": gps["latitude"],
                    "lon": gps["longitude"],
                    "temperature": env.get("temperature") if env else None,
                    "humidity": env.get("humidity") if env else None,
                    "lux": env.get("lux") if env else None,
                    "ia1_ms": None if env is None else env["_received_ms"],
                    "ia1_delta_ms": delta,
                    "quality": "ok" if env else "missing_env",
                })
            self.pending_st1 = keep
        return ready

    def _nearest(self, gps_ms: int):
        best = None
        best_delta = None
        for ia1_ms, payload in self.ia1_buffer:
            delta = abs(ia1_ms - gps_ms)
            if best_delta is None or delta < best_delta:
                best = dict(payload, _received_ms=ia1_ms)
                best_delta = delta
        if best is None or best_delta > 2500:

def build_map(points, metric, cols=60, rows=60, max_dist=80.0):

        p for p in points

    ]

    lat0 = sum(p["lat"] for p in valid) / len(valid)

    min_x = min(p[0] for p in xy)
    max_x = max(p[0] for p in xy)

    max_y = max(p[1] for p in xy)

        for col in range(cols):
            x = min_x + (max_x - min_x) * col / (cols - 1)
            y = min_y + (max_y - min_y) * row / (rows - 1)
            num = 0.0
            den = 0.0
            nearest = 1e30
            for px, py, value in xy:

                line.append(None)
            else:
                line.append(num / den)
        grid.append(line)

        "grid": grid,
        "min": min(p[2] for p in xy),
        "max": max(p[2] for p in xy),
        "avg": sum(p[2] for p in xy) / len(xy),

  const key = `${mapState.runId}:${metric}`;
  let data = mapState.cache.get(key);
  if (!data) {
    data = await request(
      `/api/map?run_id=${mapState.runId}&metric=${metric}`
    );
    mapState.cache.set(key, data);
  }
  drawMap(data);
  drawLegend(data);
  drawStats(data);
}
